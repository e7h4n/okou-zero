cmake_minimum_required(VERSION 3.24)
set(OKOU_ZERO_BUILD_VERSION "0.0.0" CACHE STRING "Package and application version")
if(NOT OKOU_ZERO_BUILD_VERSION MATCHES "^(0|[1-9][0-9]*)\\.(0|[1-9][0-9]*)\\.(0|[1-9][0-9]*)$")
  message(FATAL_ERROR "Invalid Okou Zero version")
endif()
project(okou_zero VERSION ${OKOU_ZERO_BUILD_VERSION} LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(OKOU_ZERO_DESKTOP "Build the SDL2 desktop simulator" ON)
option(OKOU_ZERO_TESTS "Build protocol and UI tests" ON)
set(OKOU_ZERO_OAUTH_CLIENT_ID "" CACHE STRING "Public OAuth client ID (never a secret)")
if(NOT OKOU_ZERO_OAUTH_CLIENT_ID MATCHES "^[A-Za-z0-9_-]*$")
  message(FATAL_ERROR "Invalid public OAuth client ID")
endif()
# One standard package targeting the production API.
set(OKOU_ZERO_API_ORIGIN "https://api.okou.ai")
configure_file(config/build_config.hpp.in generated/okou/build_config.hpp @ONLY)
include(FetchContent)
find_package(Threads REQUIRED)
find_package(CURL 7.85 REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(ZLIB REQUIRED)
find_package(nlohmann_json 3.10 REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(ALSA REQUIRED IMPORTED_TARGET alsa)
pkg_check_modules(PULSE REQUIRED IMPORTED_TARGET libpulse-simple)
find_package(Freetype REQUIRED)

set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(USE_TLS ON CACHE BOOL "" FORCE)
set(USE_OPEN_SSL ON CACHE BOOL "" FORCE)
set(USE_ZLIB ON CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
FetchContent_Declare(ixwebsocket
  URL https://codeload.github.com/machinezone/IXWebSocket/tar.gz/2efe037c9cc96fd536774f17bdb5215161ee5087
  URL_HASH SHA256=df061a0a255c938b680d4c738e2b3da74ee9dd5340bf636118061e1e0092170d
  DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_GetProperties(ixwebsocket)
if(NOT ixwebsocket_POPULATED)
  FetchContent_Populate(ixwebsocket)
  add_subdirectory(${ixwebsocket_SOURCE_DIR} ${ixwebsocket_BINARY_DIR} EXCLUDE_FROM_ALL)
endif()

add_library(okou_core STATIC src/diagnostics.cpp src/http.cpp src/auth.cpp src/config.cpp src/client.cpp src/audio.cpp src/realtime.cpp src/update_process.cpp)
target_include_directories(okou_core PUBLIC include "${CMAKE_CURRENT_BINARY_DIR}/generated" PRIVATE src)
target_compile_definitions(okou_core PUBLIC OKOU_ZERO_VERSION="${PROJECT_VERSION}")
target_link_libraries(okou_core PUBLIC nlohmann_json::nlohmann_json Threads::Threads
  CURL::libcurl OpenSSL::Crypto ixwebsocket PkgConfig::ALSA PkgConfig::PULSE)
target_compile_options(okou_core PRIVATE -Wall -Wextra -Wpedantic)

set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/config/lv_conf.h" CACHE PATH "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
FetchContent_Declare(lvgl
  URL https://codeload.github.com/lvgl/lvgl/tar.gz/85aa60d18b3d5e5588d7b247abf90198f07c8a63
  URL_HASH SHA256=b32747cf6568d86a5cb9d30f53970bec1c7a322cb624c4158d4b0e348324146b
  DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_GetProperties(lvgl)
if(NOT lvgl_POPULATED)
  FetchContent_Populate(lvgl)
  add_subdirectory(${lvgl_SOURCE_DIR} ${lvgl_BINARY_DIR} EXCLUDE_FROM_ALL)
endif()
target_link_libraries(lvgl PUBLIC Freetype::Freetype)
FetchContent_Declare(md4c
  URL https://codeload.github.com/mity/md4c/tar.gz/refs/tags/release-0.5.2
  URL_HASH SHA256=55d0111d48fb11883aaee91465e642b8b640775a4d6993c2d0e7a8092758ef21
  DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_GetProperties(md4c)
if(NOT md4c_POPULATED)
  FetchContent_Populate(md4c)
endif()
add_library(zero_md4c STATIC ${md4c_SOURCE_DIR}/src/md4c.c ${md4c_SOURCE_DIR}/src/entity.c)
target_include_directories(zero_md4c PUBLIC ${md4c_SOURCE_DIR}/src)
install(FILES ${md4c_SOURCE_DIR}/LICENSE.md DESTINATION share/doc/okou-zero RENAME MD4C-LICENSE.md)
add_executable(okou-zero src/volume_control.cpp src/markdown.cpp src/main.cpp src/display.cpp src/input.cpp src/ui.cpp src/orbit.cpp)
target_include_directories(okou-zero PRIVATE include)
target_link_libraries(okou-zero PRIVATE okou_core lvgl::lvgl zero_md4c)
target_compile_options(okou-zero PRIVATE -Wall -Wextra -Wpedantic)
target_compile_definitions(okou-zero PRIVATE OKOU_ZERO_VERSION="${PROJECT_VERSION}"
  OKOU_ZERO_DESKTOP=$<BOOL:${OKOU_ZERO_DESKTOP}>)
if(OKOU_ZERO_DESKTOP)
  find_package(SDL2 REQUIRED)
  target_link_libraries(okou-zero PRIVATE SDL2::SDL2)
endif()

if(OKOU_ZERO_TESTS)
  enable_testing()
  foreach(name auth http client audio realtime config diagnostics)
    add_executable(${name}_test tests/${name}_test.cpp)
    target_link_libraries(${name}_test PRIVATE okou_core)
    target_compile_options(${name}_test PRIVATE -Wall -Wextra -Wpedantic)
    add_test(NAME ${name} COMMAND ${name}_test)
    set_tests_properties(${name} PROPERTIES TIMEOUT 60)
  endforeach()
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME updater COMMAND ${Python3_EXECUTABLE} -m unittest discover -s ${CMAKE_CURRENT_SOURCE_DIR}/tests -p test_updater.py)
  add_test(NAME release_version COMMAND ${Python3_EXECUTABLE} -m unittest discover -s ${CMAKE_CURRENT_SOURCE_DIR}/tests -p test_release.py)
  set_tests_properties(release_version PROPERTIES TIMEOUT 15 ENVIRONMENT "PYTHONDONTWRITEBYTECODE=1")
  set_tests_properties(updater PROPERTIES TIMEOUT 60 ENVIRONMENT "PYTHONDONTWRITEBYTECODE=1")
  add_executable(update_shortcut_test tests/update_shortcut_test.cpp)
  target_include_directories(update_shortcut_test PRIVATE src)
  target_link_libraries(update_shortcut_test PRIVATE okou_core)
  add_test(NAME update_shortcut COMMAND update_shortcut_test)
  set_tests_properties(update_shortcut PROPERTIES TIMEOUT 10)
  add_executable(input_test tests/input_test.cpp src/input.cpp)
  target_include_directories(input_test PRIVATE include)
  target_compile_definitions(input_test PRIVATE OKOU_ZERO_DESKTOP=0)
  target_link_libraries(input_test PRIVATE lvgl::lvgl okou_core)
  add_test(NAME keyboard COMMAND input_test)
  add_executable(orbit_test tests/orbit_test.cpp src/orbit.cpp)
  target_include_directories(orbit_test PRIVATE include src)
  target_link_libraries(orbit_test PRIVATE lvgl::lvgl)
  add_test(NAME orbit_motion COMMAND orbit_test)
  add_executable(markdown_test tests/markdown_test.cpp src/markdown.cpp)
  target_include_directories(markdown_test PRIVATE include)
  target_link_libraries(markdown_test PRIVATE lvgl::lvgl zero_md4c)
  add_test(NAME markdown COMMAND markdown_test)
  add_executable(chat_navigation_test tests/chat_navigation_test.cpp)
  target_include_directories(chat_navigation_test PRIVATE include)
  add_test(NAME chat_navigation COMMAND chat_navigation_test)
  add_executable(framebuffer_test tests/framebuffer_test.cpp)
  target_include_directories(framebuffer_test PRIVATE include)
  target_link_libraries(framebuffer_test PRIVATE lvgl::lvgl)
  add_test(NAME framebuffer COMMAND framebuffer_test)
  set_tests_properties(framebuffer PROPERTIES TIMEOUT 10)
  add_test(NAME ui_headless COMMAND okou-zero --headless --self-test-ui --output ${CMAKE_CURRENT_BINARY_DIR}/ui-headless.ppm)
  set_tests_properties(ui_headless PROPERTIES TIMEOUT 20)
  add_test(NAME ui_update_headless COMMAND okou-zero --headless --demo-screen update --output ${CMAKE_CURRENT_BINARY_DIR}/ui-update.ppm)
  set_tests_properties(ui_update_headless PROPERTIES TIMEOUT 20)
  if(OKOU_ZERO_DESKTOP)
    add_test(NAME ui_keyboard COMMAND okou-zero --self-test-ui --screenshot ${CMAKE_CURRENT_BINARY_DIR}/ui-keyboard.bmp)
    set_tests_properties(ui_keyboard PROPERTIES ENVIRONMENT "SDL_VIDEODRIVER=dummy" TIMEOUT 20)
    foreach(screen welcome login empty chat history recording playback paused sending sent uncertain saved offline update)
      add_test(NAME ui_${screen} COMMAND okou-zero --demo-screen ${screen} --screenshot ${CMAKE_CURRENT_BINARY_DIR}/ui-${screen}.bmp)
      set_tests_properties(ui_${screen} PROPERTIES ENVIRONMENT "SDL_VIDEODRIVER=dummy" TIMEOUT 20)
    endforeach()
  endif()
endif()

  RENAME okou-zero-update PERMISSIONS OWNER_READ GROUP_READ)

install(FILES assets/okou-zero.png DESTINATION share/APPLaunch/share/images)

  DESTINATION share/doc/okou-zero RENAME Montserrat-OFL.txt)
