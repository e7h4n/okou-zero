cmake_minimum_required(VERSION 3.28)
project(okou-zero VERSION 0.1.4 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(FetchContent)
set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/config/lv_conf.h" CACHE PATH "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
FetchContent_Declare(lvgl EXCLUDE_FROM_ALL GIT_REPOSITORY https://github.com/lvgl/lvgl.git
  GIT_TAG 85aa60d18b3d5e5588d7b247abf90198f07c8a63)
FetchContent_MakeAvailable(lvgl)
add_library(okou-ui src/ui.cpp src/update_process.cpp)
target_include_directories(okou-ui PUBLIC src)
target_link_libraries(okou-ui PUBLIC lvgl::lvgl)
target_compile_options(okou-ui PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(okou-zero src/main.cpp)
target_link_libraries(okou-zero PRIVATE okou-ui)
target_compile_options(okou-zero PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_compile_definitions(okou-zero PRIVATE OKOU_VERSION="${PROJECT_VERSION}")
include(CTest)
if(BUILD_TESTING)
  add_executable(okou-tests tests/unit.cpp)
  target_link_libraries(okou-tests PRIVATE okou-ui)
  add_test(NAME model COMMAND okou-tests)
  set_tests_properties(model PROPERTIES TIMEOUT 10)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME headless-ui COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/ui.py $<TARGET_FILE:okou-zero>)
  set_tests_properties(headless-ui PROPERTIES TIMEOUT 30)
  add_test(NAME updater COMMAND ${Python3_EXECUTABLE} -m unittest discover -s ${CMAKE_CURRENT_SOURCE_DIR}/tests -p test_updater.py)
  set_tests_properties(updater PROPERTIES TIMEOUT 60 ENVIRONMENT "PYTHONDONTWRITEBYTECODE=1")
endif()
install(TARGETS okou-zero RUNTIME DESTINATION bin)
install(PROGRAMS updater/updater.py DESTINATION lib/okou-zero)
install(PROGRAMS updater/install_helper.py DESTINATION libexec RENAME okou-zero-update-install)
install(FILES updater/security.py DESTINATION lib/okou-zero)
install(FILES assets/update-public.pem DESTINATION share/okou-zero)
install(FILES packaging/okou-zero-update.sudoers DESTINATION /etc/sudoers.d RENAME okou-zero-update PERMISSIONS OWNER_READ GROUP_READ)
set(CPACK_DEBIAN_PACKAGE_DEPENDS "python3, openssl, sudo")
install(FILES packaging/okou-zero.desktop DESTINATION share/APPLaunch/applications)
install(FILES packaging/okou-zero-linux.desktop DESTINATION share/applications RENAME okou-zero.desktop)
install(FILES assets/okou-zero.png DESTINATION share/APPLaunch/share/images)
install(FILES assets/okou-zero-96.png DESTINATION share/icons/hicolor/96x96/apps RENAME okou-zero.png)
install(FILES LICENSE THIRD_PARTY_NOTICES.md DESTINATION share/doc/okou-zero)
install(FILES ${lvgl_SOURCE_DIR}/LICENCE.txt DESTINATION share/doc/okou-zero RENAME LVGL-LICENSE.txt)
install(FILES ${lvgl_SOURCE_DIR}/scripts/built_in_font/font_license/Montserrat/OFL.txt DESTINATION share/doc/okou-zero RENAME Montserrat-OFL.txt)
set(CPACK_GENERATOR DEB)
set(CPACK_PACKAGE_NAME okou-zero)
set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
set(CPACK_PACKAGE_CONTACT "Ethan Zhang <ethan@okou.ai>")
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "Keyboard-first Okou Hello World for Cardputer Zero")
if(CMAKE_SYSTEM_PROCESSOR MATCHES "aarch64|arm64")
  set(CPACK_DEBIAN_PACKAGE_ARCHITECTURE arm64)
else()
  set(CPACK_DEBIAN_PACKAGE_ARCHITECTURE amd64)
endif()
set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS ON)
set(CPACK_DEBIAN_PACKAGE_SECTION utils)
set(CPACK_PACKAGING_INSTALL_PREFIX /usr)
set(CPACK_PACKAGE_FILE_NAME "okou-zero_${PROJECT_VERSION}_${CPACK_DEBIAN_PACKAGE_ARCHITECTURE}")
include(CPack)
